cmake_minimum_required(VERSION 3.24)
project(bonsai_native LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES OFF)
find_package(CUDAToolkit REQUIRED)
find_package(OpenSSL REQUIRED)
set(LLAMA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../llama" CACHE PATH "Prism llama source")
set(LLAMA_BUILD "${CMAKE_CURRENT_SOURCE_DIR}/../../build/llama" CACHE PATH "Prism llama build")
set(ENGINE_SOURCES src/joint_five.cu src/bmega.cu src/engine.cu src/dspark_native.cu src/dspark_kernels.cu src/dspark_quant.cu src/packed1.cu experiments/binary-bitserial/bitserial.cu experiments/binary-a8-fp4/digits.cu experiments/binary-whole-map/signed_digits.cu experiments/binary-whole-map/whole_ffn.cu experiments/binary-native-sf/native.cu experiments/binary-composition/composition.cu experiments/binary-fp4-pair/pair.cu experiments/b32-imma/b32.cu src/graph_memory.cu src/kernels.cu src/fp4.cu src/sk.cu src/draft.cu src/sampling_gpu.cu src/f02-producer.cu src/f02-continuation.cu src/gguf.cpp)
add_library(engine STATIC ${ENGINE_SOURCES})
target_include_directories(engine PRIVATE "${LLAMA_ROOT}/vendor")
target_compile_definitions(engine PUBLIC NOMINMAX _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
target_compile_options(engine PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-O3;--expt-relaxed-constexpr;--generate-code=arch=compute_120a,code=sm_120a;-Xcompiler=/O2,/EHsc,/bigobj>" "$<$<COMPILE_LANGUAGE:CXX>:/O2;/EHsc;/bigobj>")
if(H5_SELECTED)
  target_compile_definitions(engine PUBLIC H5_PROFILE_GEMM H5_PDL_EARLY_GEMM_RELEASE H5_PDL_GDN_STATE_OVERLAP H5_NO_L2PF H5_PROFILE_SCALE_CARRIER H5_WEIGHT_TMA_CACHE_POLICY=2)
endif()
target_link_libraries(engine PUBLIC CUDA::cudart CUDA::cublas CUDA::cuda_driver)
set(SERVER_SOURCES src/server.cpp src/vision.cpp "${LLAMA_ROOT}/vendor/cpp-httplib/httplib.cpp" )
add_executable(h5-server ${SERVER_SOURCES})
add_executable(h5 src/main.cpp)
add_executable(refdump tools/refdump.cpp)
foreach(t h5-server h5 refdump)
  target_include_directories(${t} PRIVATE "${LLAMA_ROOT}/include" "${LLAMA_ROOT}/ggml/include" "${LLAMA_ROOT}/tools/mtmd" "${LLAMA_ROOT}/vendor" "${LLAMA_ROOT}/vendor/cpp-httplib" "${CUDAToolkit_INCLUDE_DIRS}")
  target_compile_definitions(${t} PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES CPPHTTPLIB_OPENSSL_SUPPORT)
  target_compile_options(${t} PRIVATE /O2 /EHsc /bigobj)
  target_link_directories(${t} PRIVATE "${LLAMA_BUILD}/bin" "${LLAMA_BUILD}/src" "${LLAMA_BUILD}/ggml/src" "${LLAMA_BUILD}/tools/mtmd")
  target_link_libraries(${t} PRIVATE llama ggml ggml-base)
endforeach()
target_link_libraries(h5 PRIVATE engine)
target_link_libraries(h5-server PRIVATE engine mtmd OpenSSL::SSL OpenSSL::Crypto ws2_32 crypt32)

add_executable(p12-check bench/p12_check.cu src/joint_five.cu)
target_include_directories(p12-check PRIVATE src)
target_compile_definitions(p12-check PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(p12-check PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-O3;--expt-relaxed-constexpr;--generate-code=arch=compute_120a,code=sm_120a;-Xcompiler=/O2,/EHsc>")
target_link_libraries(p12-check PRIVATE CUDA::cudart)
