cmake_minimum_required(VERSION 3.24)
project(private_gu_dict8 LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES OFF)
find_package(CUDAToolkit REQUIRED)
set(OWNER "C:/work/models")
file(GLOB OBJECTS "${CMAKE_CURRENT_SOURCE_DIR}/inherited/*.obj")
list(LENGTH OBJECTS NOBJ)
if(NOT NOBJ EQUAL 12)
 message(FATAL_ERROR "Expected12 exact frozen unchanged objects excluding engine and original dispatch; wide dispatch retained")
endif()
add_library(quality_custom SHARED src/engine.cu src/dict8.cu src/dict8_checks.cu ${OBJECTS})
target_include_directories(quality_custom PRIVATE src "${OWNER}/source/llama/vendor" "${OWNER}/source/llama/include" "${OWNER}/source/llama/ggml/include")
target_compile_definitions(quality_custom PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES H5_PROFILE_GEMM H5_PDL_EARLY_GEMM_RELEASE H5_PDL_GDN_STATE_OVERLAP H5_NO_L2PF H5_PROFILE_SCALE_CARRIER H5_WEIGHT_TMA_CACHE_POLICY=2)
target_compile_options(quality_custom PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-O3;--expt-relaxed-constexpr;--generate-code=arch=compute_120a,code=sm_120a;-lineinfo;-Xcompiler=/O2,/EHsc,/bigobj>")
set_source_files_properties(src/dict8.cu src/dict8_checks.cu PROPERTIES COMPILE_OPTIONS "-Xptxas=-v")
target_link_directories(quality_custom PRIVATE "${OWNER}/build/llama/src" "${OWNER}/build/llama/bin" "${OWNER}/build/llama/ggml/src")
target_link_libraries(quality_custom PRIVATE CUDA::cudart CUDA::cublas llama ggml ggml-base)
