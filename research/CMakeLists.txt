cmake_minimum_required(VERSION 3.28)
project(bonsai_research LANGUAGES C CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES OFF)
set(RESEARCH_VARIANT "" CACHE STRING "Explicit independent ternary research source variant")
set(variants fullk-gu packed-tail h1-graph-v3 dict8-v2 dict8-offline-identity dict8-joint-consumer)
if(NOT RESEARCH_VARIANT IN_LIST variants)
  message(FATAL_ERROR "Select RESEARCH_VARIANT from: ${variants}")
endif()
set(LLAMA_ROOT "" CACHE PATH "Explicit separately supplied PrismML llama.cpp source")
if(NOT EXISTS "${LLAMA_ROOT}/CMakeLists.txt")
  message(FATAL_ERROR "LLAMA_ROOT must name a separately supplied PrismML source tree")
endif()
find_package(CUDAToolkit REQUIRED)
find_package(OpenSSL REQUIRED)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_ROOT}" "${CMAKE_BINARY_DIR}/prism" EXCLUDE_FROM_ALL)
set(s "${CMAKE_CURRENT_SOURCE_DIR}/engines/${RESEARCH_VARIANT}/src")
set(e "${CMAKE_CURRENT_SOURCE_DIR}/external-serving")
set(core engine.cu graph_memory.cu kernels.cu fp4.cu sk.cu draft.cu sampling_gpu.cu f02-producer.cu f02-continuation.cu gguf.cpp)
set(files)
foreach(f IN LISTS core)
  list(APPEND files "${s}/${f}")
endforeach()
# These are independent exported translation units when present, not included CUDA fragments.
foreach(f fullk_gu.cu fullk_ring.cu fullk_tail.cu dict8.cu dict8_checks.cu)
  if(EXISTS "${s}/${f}")
    list(APPEND files "${s}/${f}")
  endif()
endforeach()
add_library(quality_research SHARED ${files} "${e}/api.cpp" "${e}/dispatch.cu" "${e}/gdn_q6.cu")
target_include_directories(quality_research PRIVATE "${s}" "${e}" "${CMAKE_CURRENT_SOURCE_DIR}/../engine/src" "${LLAMA_ROOT}/include" "${LLAMA_ROOT}/vendor")
target_compile_definitions(quality_research PRIVATE H5_PROFILE_GEMM H5_PDL_EARLY_GEMM_RELEASE H5_PDL_GDN_STATE_OVERLAP H5_NO_L2PF H5_PROFILE_SCALE_CARRIER H5_WEIGHT_TMA_CACHE_POLICY=2 $<$<PLATFORM_ID:Windows>:NOMINMAX _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES>)
# Frozen dispatch overrides these four public entry points; compile the original kernel bodies under their explicit names.
set_source_files_properties("${s}/kernels.cu" PROPERTIES COMPILE_DEFINITIONS "launch_gdn=qc_original_launch_gdn;launch_gdn_single=qc_original_launch_gdn_single;launch_attn_pre=qc_original_launch_attn_pre;launch_attn_tc=qc_original_launch_attn_tc")
if(EXISTS "${s}/fullk_gu.cu")
  set_source_files_properties("${s}/fullk_gu.cu" PROPERTIES COMPILE_OPTIONS "--fmad=false")
endif()
target_compile_options(quality_research PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-O3;--expt-relaxed-constexpr;--generate-code=arch=compute_120a,code=sm_120a;-lineinfo>" "$<$<AND:$<COMPILE_LANGUAGE:CUDA>,$<CXX_COMPILER_ID:MSVC>>:-Xcompiler=/O2,/EHsc,/bigobj>" "$<$<AND:$<COMPILE_LANGUAGE:CUDA>,$<NOT:$<CXX_COMPILER_ID:MSVC>>>:-Xcompiler=-O3>")
target_link_libraries(quality_research PRIVATE CUDA::cudart CUDA::cublas llama ggml ggml-base OpenSSL::Crypto)
if(WIN32)
  target_link_libraries(quality_research PRIVATE bcrypt)
endif()
